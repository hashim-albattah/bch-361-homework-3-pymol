# midpoint dot + label via Python
python

# 5) base identity (A/G/C/T/U): auto-detect + label
python
from pymol import cmd
base_map = {'A':'Adenine','G':'Guanine','C':'Cytosine','U':'Uracil','T':'Thymine'}
base_code = None
for k in base_map:
    if cmd.count_atoms(f"nuc and resn {k}"):
        base_code = k; break
cmd.delete("pa_base")
if base_code:
    sel = f"nuc and resn {base_code} and not name H*"
    com = cmd.centerofmass(sel)
    cmd.pseudoatom("pa_base", pos=com)
    cmd.show("spheres", "pa_base")
    cmd.set("sphere_scale", 0.25, "pa_base")
    cmd.color("black", "pa_base")
    cmd.set("label_font_id", 7); cmd.set("label_size", 22)
    cmd.set("label_color", "black"); cmd.set("label_outline_color", "white")
    cmd.label("pa_base", f'"Base: {base_map[base_code]} ({base_code})"')
else:
    print("Base not detected by resn (A/G/C/T/U).")
python end

# 6) DNA vs RNA: detect O2′ presence and label
python
from pymol import cmd
is_rna = cmd.count_atoms("nuc and (name O2' or name O2*)") > 0
cmd.delete("pa_type")
com2 = cmd.centerofmass("nuc and (name C1' or name C1* or name C2' or name C2* or name C3' or name C3* or name C4' or name C4* or name C5' or name C5*)")
cmd.pseudoatom("pa_type", pos=com2)
cmd.show("spheres", "pa_type")
cmd.set("sphere_scale", 0.25, "pa_type")
cmd.color("black", "pa_type")
cmd.set("label_font_id", 7); cmd.set("label_size", 22)
cmd.set("label_color", "black"); cmd.set("label_outline_color", "white")
cmd.label("pa_type", '"Type: RNA"' if is_rna else '"Type: DNA"')
print("Detected nucleoside type:", "RNA" if is_rna else "DNA")
python end
